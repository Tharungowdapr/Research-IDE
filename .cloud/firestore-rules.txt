Firestore Security Rules & Configuration
Copy these rules to Firebase Console > Firestore > Rules
"""

# ============================================================================
# FIRESTORE SECURITY RULES
# ============================================================================
# Location: Firebase Console > Firestore > Rules
# Deploy via: gcloud firestore indexes composite create --config=firestore.indexes.yaml

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // ========================================================================
    // USERS COLLECTION
    // ========================================================================
    match /users/{userId} {
      // Users can only read/write their own documents
      allow read, write: if request.auth.uid == userId;
      
      // User settings subcollection
      match /settings/{document=**} {
        allow read, write: if request.auth.uid == userId;
      }
    }
    
    // ========================================================================
    // PROJECTS COLLECTION (owned by users)
    // ========================================================================
    match /projects/{projectId} {
      // Project owner can read/write
      allow read, write: if resource.data.user_id == request.auth.uid;
      
      // Create new projects
      allow create: if request.auth != null && 
                       request.resource.data.user_id == request.auth.uid;
      
      // Subcollections (papers, gaps, ideas, etc)
      match /{document=**} {
        allow read, write: if get(/databases/$(database)/documents/projects/$(projectId)).data.user_id == request.auth.uid;
      }
    }
    
    // ========================================================================
    // PAPERS COLLECTION (linked to projects)
    // ========================================================================
    match /papers/{paperId} {
      // User can access if they own the project
      allow read, write: if get(/databases/$(database)/documents/projects/$(resource.data.project_id)).data.user_id == request.auth.uid;
      
      // Metadata subcollection
      match /metadata/{document=**} {
        allow read, write: if get(/databases/$(database)/documents/projects/$(resource.data.project_id)).data.user_id == request.auth.uid;
      }
    }
    
    // ========================================================================
    // SYSTEM COLLECTIONS (read-only for authenticated users)
    // ========================================================================
    match /system/{document=**} {
      allow read: if request.auth != null;
      allow write: if false;
    }
    
    // Deny all by default
    match /{document=**} {
      allow read, write: if false;
    }
  }
}

# ============================================================================
# FIRESTORE INDEXES (Composite)
# ============================================================================
# Deploy via: gcloud firestore indexes composite create --config=firestore.indexes.yaml

indexes:
  - collectionGroup: projects
    queryScope: COLLECTION
    fields:
      - fieldPath: user_id
        order: ASCENDING
      - fieldPath: created_at
        order: DESCENDING
  
  - collectionGroup: papers
    queryScope: COLLECTION
    fields:
      - fieldPath: project_id
        order: ASCENDING
      - fieldPath: added_at
        order: DESCENDING
  
  - collectionGroup: projects
    queryScope: COLLECTION
    fields:
      - fieldPath: user_id
        order: ASCENDING
      - fieldPath: status
        order: ASCENDING
      - fieldPath: updated_at
        order: DESCENDING
